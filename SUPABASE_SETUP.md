# Samuhik — Deployment & Setup Guide

This covers everything needed to go from the local demo to a live Netlify deployment with Supabase, Gemini AI, and POS sync wired up.

---

|---|---|

| Supabase project | supabase.com |

## Part 2 — Supabase Database Setup

### 2.1 Create the project
1. Go to [supabase.com](https://supabase.com) and create a new project.
2. **Settings → API**: Copy your **Project URL** and **anon key** (safe for client) and **service_role key** (server only — keep secret).

### 2.2 Run the schema SQL

In the Supabase **SQL Editor**, run this entire block:

```sql
-- ═══════════════════════════════════════════════════
--  1. CORE TABLES
-- ═══════════════════════════════════════════════════

CREATE TABLE IF NOT EXISTS tenants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  store_context_prompt TEXT DEFAULT '',
  web_search_enabled BOOLEAN DEFAULT false,
  enabled_platforms TEXT[] DEFAULT '{whatsapp,instagram,messenger}',
  pos_webhook_url TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  supabase_uid UUID UNIQUE,
  tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
  email TEXT NOT NULL,
  name TEXT,
  role TEXT NOT NULL CHECK (role IN ('platform_owner', 'super_admin', 'admin', 'agent', 'customer')),
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS contacts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  phone TEXT,
  email TEXT,
  metadata JSONB DEFAULT '{}',
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS conversations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
  contact_id UUID REFERENCES contacts(id) ON DELETE CASCADE,
  platform TEXT NOT NULL DEFAULT 'whatsapp',
  ai_enabled BOOLEAN DEFAULT true,
  tags TEXT[] DEFAULT '{}',
  unread_count INTEGER DEFAULT 0,
  last_message_at TIMESTAMPTZ DEFAULT now(),
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
  conversation_id UUID REFERENCES conversations(id) ON DELETE CASCADE,
  direction TEXT NOT NULL CHECK (direction IN ('inbound', 'outbound')),
  sender_type TEXT NOT NULL CHECK (sender_type IN ('customer', 'ai', 'agent')),
  content TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS inventory (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
  sku TEXT NOT NULL,
  name TEXT NOT NULL,
  price NUMERIC(10, 2) NOT NULL DEFAULT 0,
  stock INTEGER NOT NULL DEFAULT 0,
  category TEXT,
  updated_at TIMESTAMPTZ DEFAULT now(),
  created_at TIMESTAMPTZ DEFAULT now(),
  UNIQUE(tenant_id, sku)
);

-- ═══════════════════════════════════════════════════
--  2. FIRST-USER TRIGGER
--  The first signup becomes Platform Owner.
--  Every subsequent signup becomes a store Admin with their own tenant.
-- ═══════════════════════════════════════════════════

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger AS $$
DECLARE
  is_first BOOLEAN;
  new_tenant_id UUID;
BEGIN
  SELECT NOT EXISTS (SELECT 1 FROM public.users) INTO is_first;

  IF is_first THEN
    INSERT INTO public.users (supabase_uid, email, name, role)
    VALUES (new.id, new.email, COALESCE(new.raw_user_meta_data->>'full_name', 'Platform Owner'), 'platform_owner');
  ELSE
    INSERT INTO public.tenants (name, slug)
    VALUES (
      COALESCE(new.raw_user_meta_data->>'full_name', 'My Store'),
      lower(replace(COALESCE(new.raw_user_meta_data->>'full_name', 'shop'), ' ', '-'))
        || '-' || floor(random() * 100000)::text
    )
    RETURNING id INTO new_tenant_id;

    INSERT INTO public.users (supabase_uid, email, name, role, tenant_id)
    VALUES (new.id, new.email, COALESCE(new.raw_user_meta_data->>'full_name', 'User'), 'admin', new_tenant_id);
  END IF;

  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- ═══════════════════════════════════════════════════
--  3. ROW LEVEL SECURITY (RLS)
-- ═══════════════════════════════════════════════════

ALTER TABLE tenants ENABLE ROW LEVEL SECURITY;
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE contacts ENABLE ROW LEVEL SECURITY;
ALTER TABLE conversations ENABLE ROW LEVEL SECURITY;
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;
ALTER TABLE inventory ENABLE ROW LEVEL SECURITY;

-- Helper: is the current user a platform-level admin?
CREATE OR REPLACE FUNCTION is_platform_admin()
RETURNS BOOLEAN AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.users
    WHERE supabase_uid = auth.uid()
    AND role IN ('platform_owner', 'super_admin')
  );
$$ LANGUAGE sql SECURITY DEFINER;

-- Helper: get the current user's tenant_id
CREATE OR REPLACE FUNCTION my_tenant_id()
RETURNS UUID AS $$
  SELECT tenant_id FROM public.users WHERE supabase_uid = auth.uid() LIMIT 1;
$$ LANGUAGE sql SECURITY DEFINER;

CREATE POLICY "conversations_access" ON conversations FOR ALL USING (
  tenant_id = my_tenant_id() OR is_platform_admin()
);
CREATE POLICY "messages_access" ON messages FOR ALL USING (
  tenant_id = my_tenant_id() OR is_platform_admin()
);
CREATE POLICY "inventory_access" ON inventory FOR ALL USING (
  tenant_id = my_tenant_id() OR is_platform_admin()
);
CREATE POLICY "contacts_access" ON contacts FOR ALL USING (
  tenant_id = my_tenant_id() OR is_platform_admin()
);
CREATE POLICY "tenants_access" ON tenants FOR SELECT USING (
  id = my_tenant_id() OR is_platform_admin()
);
CREATE POLICY "users_self_read" ON users FOR SELECT USING (
  supabase_uid = auth.uid() OR is_platform_admin()
);
```

### 2.3 Enable Realtime
Go to **Supabase → Database → Replication** and enable for:
- `messages`
- `conversations`

## Part 3 — Netlify Deployment

### 3.1 Connect your repo
1. Push this project to a GitHub repo.
2. Go to Netlify → **Add new site → Import from Git**.
3. Select the repo. Netlify will auto-detect Vite/React.

### 3.2 Set environment variables

In **Netlify → Site → Environment Variables**, add ALL of these:

| Variable | Value | Notes |
|---|---|---|
| `VITE_DEMO_MODE` | `false` | **Critical** — without this, the app runs in demo mode |
| `VITE_SUPABASE_URL` | `https://xxx.supabase.co` | From Supabase Settings → API |
| `VITE_SUPABASE_ANON_KEY` | `eyJ...` | From Supabase Settings → API (safe to expose) |
| `GEMINI_API_KEY` | `AIza...` | From aistudio.google.com — server only |
| `SUPABASE_SERVICE_ROLE_KEY` | `eyJ...` | From Supabase Settings → API (NEVER expose to browser) |
| `SAMUHIK_POS_API_KEY` | `<random string>` | You generate this — `openssl rand -hex 32` |

> **Why two Supabase keys?**
> - `VITE_SUPABASE_ANON_KEY` is used by the frontend (public, respects RLS)
> - `SUPABASE_SERVICE_ROLE_KEY` is used by the `pos-sync` Netlify function to bypass RLS for server-side inventory writes

### 3.3 Deploy
Trigger a deploy. The `netlify.toml` already has:
- `node_bundler = "esbuild"` — fixes ESM function loading (this was the main deployment bug)
- Redirects for React Router SPA

### 3.4 First login
1. Open your deployed site.
2. Sign up with your email — the first account auto-becomes Platform Owner.
3. Every subsequent signup creates a new store tenant.

## Part 4 — POS Sync Setup

### 4.1 How it works

Samuhik uses an **inbound webhook** model: your POS system POSTs inventory data to Samuhik whenever stock changes. This is simpler and more reliable than polling.

```
POS System ──POST──► /.netlify/functions/pos-sync ──upsert──► Supabase inventory
```

The function handles both push-on-change and full nightly dumps.

### 4.2 Configure your POS (PasalOS)

In PasalOS webhook settings:

| Field | Value |
|---|---|
| Webhook URL | `https://your-site.netlify.app/.netlify/functions/pos-sync` |
| Method | `POST` |
| Header key | `x-api-key` |
| Header value | *(your `SAMUHIK_POS_API_KEY`)* |
| Content-Type | `application/json` |
| Trigger | On stock change / nightly full sync |

**Payload (standard format):**
```json
{
  "tenant_id": "your-tenant-uuid-from-samuhik-dashboard",
  "items": [
    { "sku": "RB-3025", "name": "Ray-Ban Aviator Classic", "price": 15000, "stock": 12 },
    { "sku": "OO-9102", "name": "Oakley Holbrook", "price": 18500, "stock": 8 }
  ]
}
```

**PasalOS native fields are also accepted** (the function maps them automatically):

| PasalOS field | Samuhik equivalent |
|---|---|
| `product_code` | `sku` |
| `product_name` | `name` |
| `selling_price` | `price` |
| `quantity_on_hand` | `stock` |
| `product_category` | `category` |

### 4.3 Test the connection

From **Samuhik → Store Config → POS Integration**, click **Test Connection**. This POSTs your current inventory to the function and shows whether it's reachable and authenticated.

### 4.4 Verify in Supabase

After a successful sync, check **Supabase → Table Editor → inventory**. Rows should be upserted under your `tenant_id`.

## Part 5 — Dependencies Reference

All production dependencies are already in `package.json`. No additional installs needed for POS sync.

| Package | Used for |
|---|---|
| `@supabase/supabase-js` | Database reads (frontend) + inventory upsert (pos-sync function) |
| `react`, `react-dom` | UI |
| `react-router-dom` | Client-side routing |
| `lucide-react` | Icons |
| `vite` | Build tool |

**Built-in Node.js modules used by functions (no install needed):**
| Module | Used for |
|---|---|
| `fetch` | HTTP calls to Gemini API (Node 18+) |
| `crypto` | HMAC signature verification (future) |
| `Buffer` | Base64 decode of function body |
| `process.env` | Reading env vars |

**If you later need HMAC webhook verification** (PasalOS sends a signature):
```js
// Already supported in Node built-ins — no npm install required
import crypto from 'crypto';
const sig = crypto.createHmac('sha256', process.env.SAMUHIK_POS_API_KEY)
  .update(rawBody).digest('hex');
if (sig !== event.headers['x-signature']) return json(401, { error: 'Bad signature' });
```

## Part 6 — Known Issues Fixed

| Issue | Root cause | Fix applied |
|---|---|---|
| Netlify function returns 502 | ESM `export` syntax not bundled | Added `node_bundler = "esbuild"` to `netlify.toml` |
| App stays in demo mode on deploy | `VITE_DEMO_MODE` not set in Netlify dashboard | Must be set to `false` explicitly |
| POS sync button did nothing | `setTimeout` mock, no real function | Replaced with real `pos-sync.js` Netlify function |
| AI calls work locally but fail on Netlify | `GEMINI_API_KEY` not set in Netlify env vars | Set via Netlify dashboard (not in code) |
| Multiple logo SVG mask ID conflicts | Static IDs in reused component | Fixed with `React.useId()` per instance |

## Part 7 — Local Development Checklist

```bash
# 1. Copy env file
cp .env.example .env
# Edit .env with your real keys (or leave VITE_DEMO_MODE=true to skip Supabase)

# 2. Install dependencies
npm install

# 3. Run dev server (Netlify functions are proxied automatically)
npm run dev

# 4. Test Gemini function
curl -X POST http://localhost:5173/.netlify/functions/gemini \
  -H "Content-Type: application/json" \
  -d '{"conversation":{"messages":[{"from":"customer","text":"Hello"}],"tenantName":"Test Shop"},"inventory":[],"orders":[]}'

# 5. Test POS sync function
curl -X POST http://localhost:5173/.netlify/functions/pos-sync \
  -H "Content-Type: application/json" \
  -d '{"tenant_id":"tenant-001","items":[{"sku":"TEST-001","name":"Test Item","price":100,"stock":10}]}'
```
